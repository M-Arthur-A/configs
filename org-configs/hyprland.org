#+begin_src json :tangle "/home/arthur/Applications/wofi-vpn.sh" :mkdirp yes
#!/bin/bash

# Получаем список VPN-подключений
get_vpn_list() {
    nmcli -t -f NAME,TYPE,DEVICE connection show | grep :vpn: | while IFS=: read -r name type device; do
        # Проверяем активные подключения
        if nmcli -t -f NAME connection show --active | grep -q "^$name$"; then
            echo "🟢 $name"
        else
            echo "🔴 $name"
        fi
    done
}

# Основной скрипт
selected=$(get_vpn_list | wofi -d -p "VPN" -W 300 -H 400)

if [[ -n "$selected" ]]; then
    vpn_name=$(echo "$selected" | sed 's/^[🔴🟢] //')
    current_status=$(echo "$selected" | cut -d' ' -f1)

    if [[ "$current_status" == "🔴" ]]; then
        # Подключаемся к VPN
        nmcli connection up "$vpn_name"
        notify-send "VPN" "Подключение к $vpn_name установлено" -i network-vpn
    else
        # Отключаем VPN
        nmcli connection down "$vpn_name"
        notify-send "VPN" "Подключение $vpn_name отключено" -i network-vpn
    fi
fi
